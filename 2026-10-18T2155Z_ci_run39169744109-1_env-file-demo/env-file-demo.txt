# Part 1: a generic app started with --env-file (podman 4.9.3)

## The .env file

$ cat demo/env-file-demo/generic-app.env
# A typical .env file: every value below is a credential.
# EXAMPLE VALUES ONLY — used by scripts/env-file-demo.sh to show who can read them.
DB_HOST=db.internal
DB_USER=notes_app
DB_PASSWORD=Example-Only-Pa55w0rd
REDIS_URL=redis://:Example-Only-R3dis@cache.internal:6379
API_KEY=sk_test_EXAMPLE_ONLY_4f9a2c7e


## 1. Start it headless (-d), removed when it stops (--rm)

$ podman run -q -d --rm --name envdemo --env-file demo/env-file-demo/generic-app.env docker.io/library/alpine:3.20 sleep 600
dd1be0591cc0e09ba89530bb121f5c8a51e3cfb545afb614cfc85c7f6fd18415


## 2. Anyone who can exec into the container reads every value

$ podman exec envdemo env
container=podman
DB_HOST=db.internal
DB_USER=notes_app
DB_PASSWORD=Example-Only-Pa55w0rd
REDIS_URL=redis://:Example-Only-R3dis@cache.internal:6379
API_KEY=sk_test_EXAMPLE_ONLY_4f9a2c7e
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
HOME=/root


## 3. The same values sit in the process's environment block

$ podman exec envdemo sh -c 'tr "\0" "\n" < /proc/1/environ'
container=podman
DB_HOST=db.internal
DB_USER=notes_app
DB_PASSWORD=Example-Only-Pa55w0rd
REDIS_URL=redis://:Example-Only-R3dis@cache.internal:6379
API_KEY=sk_test_EXAMPLE_ONLY_4f9a2c7e
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
HOME=/root
HOSTNAME=dd1be0591cc0


## 4. No shell needed at all: the container runtime hands them out

$ podman inspect envdemo --format '{{range .Config.Env}}{{println .}}{{end}}'
container=podman
DB_HOST=db.internal
DB_USER=notes_app
DB_PASSWORD=Example-Only-Pa55w0rd
REDIS_URL=redis://:Example-Only-R3dis@cache.internal:6379
API_KEY=sk_test_EXAMPLE_ONLY_4f9a2c7e
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
HOME=/root
HOSTNAME=dd1be0591cc0


## 5. Every child process inherits them (a crash reporter, a debug shell, a subprocess)

$ podman exec envdemo sh -c 'sh -c env | grep -E "PASSWORD|API_KEY"'   # a process started by another process
API_KEY=sk_test_EXAMPLE_ONLY_4f9a2c7e
DB_PASSWORD=Example-Only-Pa55w0rd


## 6. On the host, the container's process is an ordinary process

$ pid=$(podman inspect envdemo --format '{{.State.Pid}}')   # 2516
$ sudo cat /proc/$pid/environ | tr '\0' '\n' | grep PASSWORD
DB_PASSWORD=Example-Only-Pa55w0rd

$ podman kill envdemo
envdemo


# Part 2: Secure Notes, given only where its secrets are

## Its env file: secret IDs and hostnames, no credentials

$ cat demo/env-file-demo/secure-notes.env
# What Secure Notes is given instead: where to find its secrets, never the secrets.
NODE_ENV=production
AWS_REGION=us-east-1
SECRETS_MODE=sdk
DB_SECRET_ID=arn:aws:secretsmanager:us-east-1:111122223333:secret:rds!db-0a1b2c3d-AbCdEf
# The Redis secret is found by the naming convention: /elasticache/prod/secure-notes/app
SECRETS_ENV=prod
REDIS_SECRET_RESOURCE=secure-notes
DB_HOST=secure-notes-db.abc123.us-east-1.rds.amazonaws.com
DB_SSL=true
REDIS_HOST=master.secure-notes-cache.abc123.use1.cache.amazonaws.com
REDIS_TLS=true
LOG_LEVEL=info


## The same lookups (command overridden to sleep: no AWS needed to look)

$ podman run -q -d --rm --name sndemo --env-file demo/env-file-demo/secure-notes.env secure-notes-app:dev sleep 600
988693ff0ad5c79efd501ac2fa6d57020d06079d29d3a01e5fcea382b367f11e

$ podman exec sndemo env | grep -Ei 'pass|secret|token|key|url'

memory only, and the app refuses to start if DB_PASSWORD, DATABASE_URL, REDIS_URL… are set.
